function whatCentury(year) //Задание 3
{//Вычисление века из года
  let century = Math.ceil(Number(year) / 100);
  
  if (century % 100 >= 11 && century % 100 <= 13)
  {
    return century + "th";
  }
  else if (century % 10 == 1)
  {
    return century + "st";
  }
  else if (century % 10 === 2)
  {
    return century + "nd";
  }
  else if (century % 10 === 3)
  {
    return century + "rd";
  }
  else
  {
    return century + "th";
  }
}
