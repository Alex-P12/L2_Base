function findMissing(list) //Задание 4
{//Функция нахождения элемента вне арифметической прогрессии
  let d = (list[list.length - 1] - list[0]) / (list.length);
  
  for (let j = 1; j < list.length; j++)
  {
    if (list[j] - list[j - 1] != d)
    {
      return list[j - 1] + d;
    }
  }
  
  return 0;
}

console.log(findMissing([1, 3, 5, 9, 11]));
